\noindent\textbf{Programme de construction 1}\par\smallskip\noindent
\begin{itemize}
    \item Construire un segment $[\mathspoint{AB}]$ sachant que
    $\mathspoint{AB}=10\text{ cm}$.
    \item Placer $\mathspoint{C}$ au milieu de $\mathspoint{AB}$.
    \item Construire la droite $(\mathspoint{(d)})$ perpendiculaire
    à $\mathspoint{[AB]}$ passant par $\mathspoint{C}$.
    \item Sur la droite $\mathspoint{(d)}$, placer les points
    $\mathspoint{D}$ et $\mathspoint{E}$ sachant que
    $\mathspoint{DE}=6\text{ cm}$ et que $\mathspoint{C}$ est le
    milieu de $[\mathspoint{DE}]$.
    \item Construire $\mathspoint{ADBE}$.
\end{itemize}
