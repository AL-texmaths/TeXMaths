        \item $\text{...... }>3\,939,01>\text{ ...... }$

    % Exercice 3
    On considère le nombre $86\,071,235$.
    \begin{enumerate}

        \item Écrire sa partie décimale.
        \item Écrire le chiffre des millièmes.
        \item Écrire le chiffre des milliers.
        \item Écrire le nombre des milliers.
        \item Écrire le rang du chiffre $3$~?
        \item Écrire le rang du chiffre $8$~?
    \end{enumerate}
